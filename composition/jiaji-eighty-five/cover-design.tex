\newcommand{\JJBCoverWordmarkLayerBegin}{%
  \begin{scope}\pgfsetfillcolor{NPEEWordmarkSolid}\pgfseteorule}
\newcommand{\JJBCoverWordmarkLayerEnd}{%
  \pgfusepath{fill}\end{scope}}
\newcommand{\JJBCoverPaintWordmark}[1]{%
  \begin{scope}
    \pgftransformshift{\pgfpointanchor{#1}{center}}%
    \pgfsetxvec{\pgfpoint{\JJBCoverWordmarkUnitX}{0pt}}%
    \pgfsetyvec{\pgfpoint{0pt}{\JJBCoverWordmarkUnitY}}%
    \begingroup
      \let\NPEECoverWordmarkLayerBegin\JJBCoverWordmarkLayerBegin
      \let\NPEECoverWordmarkLayerEnd\JJBCoverWordmarkLayerEnd
      \input{wordmark-path.tex}%
    \endgroup
  \end{scope}}

\providecommand{\NPEECoverPage}[2]{}
\renewcommand{\NPEECoverPage}[2]{%
  \thispagestyle{empty}%
  \null
  \begin{tikzpicture}[remember picture,overlay]
    \clip (current page.south west) rectangle (current page.north east);
    \JJBCoverPaintFixedPaper
    \JJBCoverPaintRibbon
    \ifNPEEAfour
      \coordinate (jjb-cover-content-center) at
        ([xshift=\JJBCoverContentShift]current page.center);
      \begin{scope}[shift={(jjb-cover-content-center)},
        scale=\JJBCoverDesignScale,transform shape]
        \node[anchor=north west,text=NPEEOriginalCoverInk,
          font=\NPEECoverTitleFont,inner sep=0pt]
          at (\NPEEAfourCoverTitleX,\NPEEAfourCoverTitleY)
          {\JJBCoverTitleWithSubject};
        \node[anchor=north west,text=NPEEOriginalCoverInk,
          font=\NPEECoverSubtitleFont,text width=191mm,align=left,inner sep=0pt]
          at (\NPEEAfourCoverSubtitleX,\NPEEAfourCoverSubtitleY)
          {\NPEECoverRenderSubtitle{#1}};
        \node[anchor=north west,text=black,
          font=\NPEECoverEditorFont,inner sep=0pt]
          at (\NPEEAfourCoverEditorX,\NPEEAfourCoverEditorY) {#2};
      \end{scope}
      \begin{scope}[shift={(current page.center)},scale=\JJBCoverDesignScale]
        \coordinate (jjb-cover-reference-wordmark-center) at (9.7mm,-89.4mm);
        \JJBCoverPaintReferenceWordmark{jjb-cover-reference-wordmark-center}
      \end{scope}
    \else
      \begin{scope}[shift={(current page.north west)},scale=\JJBCoverDesignScale]
        \node[anchor=north west,text=NPEEOriginalCoverInk,
          font=\NPEECoverTitleFont,inner sep=0pt]
          at (19mm,-66mm) {\JJBCoverTitleWithSubject};
        \node[anchor=north west,text=NPEEOriginalCoverInk,
          font=\NPEECoverSubtitleFont,text width=191mm,align=left,inner sep=0pt]
          at (19mm,-96mm) {\NPEECoverRenderSubtitle{#1}};
        \node[anchor=north west,text=black,
          font=\NPEECoverEditorFont,inner sep=0pt]
          at (20.25mm,-130mm) {#2};
      \end{scope}
      \begin{scope}[shift={(current page.center)},scale=\JJBCoverDesignScale,
        transform shape]
        \coordinate (jjb-cover-reference-wordmark-center) at (9.7mm,-89.4mm);
        \JJBCoverPaintReferenceWordmark{jjb-cover-reference-wordmark-center}
      \end{scope}
    \fi
  \end{tikzpicture}%
